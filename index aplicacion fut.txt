export default function GitLangHub() {
  const [selectedLang, setSelectedLang] = useState('en');
  const [activeTab, setActiveTab] = useState('flashcards');
  const [cardIndex, setCardIndex] = useState(0);
  const [isFlipped, setIsFlipped] = useState(false);
  const [xp, setXp] = useState(1450);
  const [streak, setStreak] = useState(12);

  // Quiz State
  const [quizIndex, setQuizIndex] = useState(0);
  const [selectedOption, setSelectedOption] = useState(null);
  const [quizScore, setQuizScore] = useState(0);
  const [quizFinished, setQuizFinished] = useState(false);

  const currentLangData = LANGUAGE_DATA[selectedLang] || LANGUAGE_DATA.en;

  const handleSpeak = (text) => {
    if ('speechSynthesis' in window) {
      window.speechSynthesis.cancel();
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.lang = currentLangData.code;
      window.speechSynthesis.speak(utterance);
    }
  };

  const handleQuizAnswer = (optionIdx) => {
    if (selectedOption !== null) return;
    setSelectedOption(optionIdx);

    const isCorrect = optionIdx === currentLangData.quiz[quizIndex].correct;
    if (isCorrect) {
      setQuizScore((prev) => prev + 1);
      setXp((prev) => prev + 50);
    }

    setTimeout(() => {
      if (quizIndex + 1 < currentLangData.quiz.length) {
        setQuizIndex((prev) => prev + 1);
        setSelectedOption(null);
      } else {
        setQuizFinished(true);
      }
    }, 1200);
  };

  const resetQuiz = () => {
    setQuizIndex(0);
    setSelectedOption(null);
    setQuizScore(0);
    setQuizFinished(false);
  };

  return (
    <div className="min-h-screen bg-[#0d1117] text-[#c9d1d9] font-sans antialiased">
      {/* Top Navbar */}
      <header className="border-b border-[#30363d] bg-[#161b22] px-6 py-4 flex flex-wrap items-center justify-between gap-4 sticky top-0 z-50">
        <div className="flex items-center gap-3">
          <div className="p-2 bg-[#238636] rounded-lg text-white">
            <Code2 size={24} />
          </div>
          <div>
            <h1 className="text-xl font-bold text-white flex items-center gap-2">
              GitLang Hub
              <span className="text-xs px-2 py-0.5 rounded-full bg-[#21262d] text-[#8b949e] border border-[#30363d]">
                v2.4.0
              </span>
            </h1>
            <p className="text-xs text-[#8b949e]">Learn languages like open-source code</p>
          </div>
        </div>

        {/* Stats & Selector */}
        <div className="flex items-center gap-6">
          <div className="flex items-center gap-4 bg-[#0d1117] px-4 py-2 rounded-md border border-[#30363d]">
            <div className="flex items-center gap-1.5 text-amber-500 font-semibold text-sm">
              <Flame size={18} className="animate-pulse" />
              <span>{streak} days</span>
            </div>
            <div className="h-4 w-[1px] bg-[#30363d]" />
            <div className="flex items-center gap-1.5 text-emerald-400 font-semibold text-sm">
              <Award size={18} />
              <span>{xp} XP</span>
            </div>
